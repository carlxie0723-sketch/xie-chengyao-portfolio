void processPushButtons()
{
    static bool buttonBeenReleased = true;

    // 使用 int 存储 read() 返回值，避免窄化转换警告
    int prevState = pushButtons.read();
    ThisThread::sleep_for(20ms); // RTOS 去抖延时
    int nextState = pushButtons.read();

    if (prevState == nextState)
    {
        // 低电平有效（Active-Low）：反转并屏蔽提取低3位
        int buttonStates = (~nextState & 0x07);

        if (buttonStates == 0x00)
        {
            buttonBeenReleased = true; // 所有按键释放后，重置锁存状态
        }
        else if (buttonBeenReleased)
        {
            buttonBeenReleased = false; // 触发后锁定，防止长按连续触发

            printf("button states: %02X\r\n", buttonStates);

            switch (buttonStates)
            {
                case 0x01:
                    printf(" change modes\r\n");
                    break;
                case 0x02:
                    printf(" decrement\r\n");
                    break;
                case 0x04:
                    printf(" increment\r\n");
                    break;
                default:
                    printf("get your fat fingers off more than 1 button\r\n");
                    break;
            }
        }
    }
}

int main()
{
    // 配置按键引脚为内部上拉模式
    pushButtons.mode(PullUp);
    int loopCounter = 0;

    while (1)
    {
        processPushButtons();

        // 利用按键检测的 20ms 周期进行计数，实现约 500ms 的 LED 翻转
        loopCounter++;
        if (loopCounter >= 25) // 25 * 20ms = 500ms
        {
            myled = !myled;
            loopCounter = 0;
        }
    }
}

int main()
{
    pushButton.mode(PullUp);

    bool lastSteadyState = true; // 上一次稳定的按键状态 (默认 PullUp 为 true)
    bool currentState = true;    // 当前读取的状态

    while (1)
    {
        currentState = pushButton.read();

        // 两次读取之间加入延时以避开抖动
        ThisThread::sleep_for(20ms);

        // 只有当当前状态与上一轮的稳定状态不一致时，才说明发生了状态改变
        if (currentState != lastSteadyState)
        {
            // 确认新状态已稳定（再次校验）
            if (pushButton.read() == currentState)
            {
                lastSteadyState = currentState; // 更新稳定状态

                // 判断是否为按下边沿 (Active-Low: false 代表按下)
                if (currentState == false)
                {
                    myled = !myled;
                    printf("LED status: %s\r\n", myled ? "ON" : "OFF");
                }
            }
        }
    }
}

针对上一轮评估中提出的问题，以下是具体的**代码修改建议与优化后的重构代码**：

**重点修改说明**

* **修正逻辑 Bug：**
* `serialIf()`：将独立的 `if` 改为 `elif`，解决输入 `1` 却触发 `else` 的问题。
* `leapyear()`：修正运算符优先级与逻辑，确保 2000 年（世纪闰年）等判断准确。


* **消除副作用 (In-place Mutation)：**
* `NumberMin()` / `NumberMax()`：改用 Python 内置的 `min()` 和 `max()`，避免 `num.sort()` 破坏原始列表顺序。


* **提高健壮性 (异常与除零保护)：**
* `enterFloat()`：预先声明 `num = None`，防止变量未定义错误。
* `NumberAve()`：增加空列表检查，防止 `ZeroDivisionError`。
* `curCalc()` / `resCalc()` / `paraRes()`：加入分母非零校验。
